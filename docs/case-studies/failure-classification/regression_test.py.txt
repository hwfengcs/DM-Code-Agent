"""同一组行为断言用于历史缺陷版本与修复版本，不调用模型。"""

import inspect

import pytest

from dm_agent.core.observation import is_failure_observation


@pytest.mark.parametrize(
    "action,observation,failed",
    [
        ("read_file", "1 | from errors import NotFound", False),
        ("edit_file", "已替换 1 处。\n2 | raise ValueError('invalid')", False),
        ("list_directory", "errors.py\nmain.py", False),
        ("run_python", "stdout:\nERROR: expected fixture\nreturncode: 0", False),
        ("run_tests", "stdout:\nall tests passed\nreturncode: 0", False),
        ("run_python", "stderr:\nprocess stopped\nreturncode: 2", True),
        ("run_shell", "stderr:\nprocess stopped\nreturncode: 255", True),
        ("read_file", "文件 missing.py 不存在。", True),
        ("read_file", "路径 tests 不是文件。", True),
        ("edit_file", "Tool execution failed: permission denied", True),
        ("edit_file", "[语法检查] 未通过：invalid syntax", True),
        ("third_party_tool", "Error: extension failed", True),
    ],
)
def test_observation_status(action, observation, failed):
    # 历史接口没有 action 参数；调用真实旧接口，避免以 TypeError 冒充缺陷复现。
    kwargs = {"action": action} if "action" in inspect.signature(is_failure_observation).parameters else {}
    assert is_failure_observation(observation, **kwargs) is failed
